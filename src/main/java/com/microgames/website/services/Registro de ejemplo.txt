@Service
@RequiredArgsConstructor
public class UserRegistrationService {

    private final UserDataRepository userRepo;
    private final AuthorityRepository authorityRepo;

    @Transactional
    public UserData registerUser(String email, String rawPassword, String bio) {

        UserData user = new UserData();
        user.setEmail(email);

        UserProfile profile = new UserProfile();
        profile.setUserData(user); // @MapsId toma el id de aquí
        profile.setBio(bio);
        user.setProfile(profile);

        UserSecurity security = new UserSecurity();
        security.setUserData(user);
        security.setPasswordHash(hash(rawPassword));
        user.setSecurity(security);

        Authority defaultRole = authorityRepo.findByName("ROLE_USER")
                .orElseThrow();
        user.getAuthorities().add(defaultRole);

        // gracias al cascade = ALL en UserData, esto persiste
        // user + profile + security en una sola transacción
        return userRepo.save(user);
    }
}
